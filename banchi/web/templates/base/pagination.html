{% macro render_pagination(url, page, page_size, size_per_page, kwargs) -%}
{% set ns = namespace(page_url=url) %} {%- for i in range(1, page_size+1) %} {% if i == page %} {{ i }} {% else %} {% set ns.delimitor = '?' %} {%- if '?' in url %} {% set ns.delimitor = '&' %} {% endif -%} {%- set ns.page_url = url + ns.delimitor + "page=" + i|string + "&size_per_page=" + size_per_page|string %} {%- if kwargs %} {% for key, value in kwargs.items() if value %} {% set ns.page_url = ns.page_url + "&" + key + "=" + value|string %} {% endfor -%} {% endif -%} {{ i }} {% endif %} {% endfor -%}
{%- endmacro %}
